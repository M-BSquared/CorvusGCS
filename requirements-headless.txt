# The runtime dependencies that browser mode (serve.py) and the test suite
# need — everything except the desktop app's Qt stack.
#
# Split out from requirements.txt on purpose: both CI pipelines run the full
# pytest suite without PyQt6, because no headless test imports Qt and pulling
# ~100 MB of Chromium into every test job to not use it is pure wall-clock.
# They used to get these three with `pip install pymavlink pyserial paramiko`
# — no version floors at all — so CI could pass against a pymavlink older
# than the >=2.4 the code requires.

# MAVLink protocol + dialects. 2.4 is the floor for the message set the bridge
# and the forwarder use.
pymavlink>=2.4
# SSH to the companion computer.
paramiko>=3.0
# The serial transport under MAVLink, and the firmware uploader's own use of
# the bootloader device.
pyserial>=3.5
